cmake_minimum_required(VERSION 3.13)
project(RopeSim LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HW8_GUI "Build the original OpenGL viewer" ON)
option(HW8_SANITIZE "Enable address/undefined behavior checks" OFF)
add_library(rope_core src/rope.cpp)
target_include_directories(rope_core PUBLIC src CGL/include)
add_executable(rope_tests src/tests.cpp)
target_link_libraries(rope_tests PRIVATE rope_core)
add_executable(rope_trace src/trace.cpp)
target_link_libraries(rope_trace PRIVATE rope_core)
if(HW8_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    add_link_options(-fsanitize=address,undefined -no-pie)
    foreach(target rope_core rope_tests rope_trace)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined -no-pie)
    endforeach()
endif()
if(HW8_GUI)
    find_package(OpenGL REQUIRED)
    find_package(Freetype REQUIRED)
    find_package(GLEW REQUIRED)
    find_package(glfw3 REQUIRED)
    file(GLOB CGL_SOURCES CONFIGURE_DEPENDS CGL/src/*.cpp CGL/src/osdfont.c)
    add_library(CGL STATIC ${CGL_SOURCES})
    target_include_directories(CGL PUBLIC CGL/include PRIVATE CGL/src)
    target_link_libraries(CGL PUBLIC OpenGL::GL Freetype::Freetype GLEW::GLEW glfw)
    add_executable(ropesim src/application.cpp src/main.cpp)
    target_link_libraries(ropesim PRIVATE rope_core CGL)
endif()
enable_testing()
add_test(NAME rope_checks COMMAND rope_tests)
