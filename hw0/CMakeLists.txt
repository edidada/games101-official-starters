cmake_minimum_required(VERSION 3.16)
project(Transformation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_executable(Transformation main.cpp)
target_link_libraries(Transformation PRIVATE Eigen3::Eigen)
enable_testing()
add_executable(test_transform test_transform.cpp)
target_link_libraries(test_transform PRIVATE Eigen3::Eigen)
add_test(NAME transform_geometry COMMAND test_transform)
